func valor(int ordem) -> int {
    return ordem * 100;
}
func saldo_disponivel(int ordem) -> int {
    return 500;
}
func limite_credito(int ordem) -> int {
    return 1000;
}
proc debita(int ordem) {
}
proc usa_limite(int ordem) {
}
proc liquida(int ordem, out int status) {
    if (saldo_disponivel(ordem) >= valor(ordem)) {
        debita(ordem);
        status := 1;
    } else if (limite_credito(ordem) >= valor(ordem)) {
        usa_limite(ordem);
        status := 2;
    } else {
        status := 0;
    }
}
proc main() {
    int status := 0;
    liquida(3, out status);
    escrever(status);
    liquida(8, out status);
    escrever(status);
    liquida(20, out status);
    escrever(status);
}
