// Coerção int -> float (critério 3A). O C gerado mostra cada coerção como ((double)x).
// Saída esperada (uma por linha): 8 3.5 2 7.5 1 4.5 1.5 1
const float TAXA := 1.5;
const float DOIS := 2;

func metade(float x) -> float {
    return x / 2;
}

func dobro(int n) -> float {
    return n * 2;
}

proc main() {
    float x := 2.5;
    float y := 1.25;
    int c := 3;
    escrever(x * x - y + c);

    float f := 1 + 2.5;
    escrever(f);

    float g := c;
    g := g - 1;
    escrever(g);

    escrever(metade(15));
    escrever(c / 2);
    escrever(dobro(c) - TAXA);

    float[2] v := [TAXA; 2];
    escrever(v[1]);

    if (c > x && DOIS = 2) {
        escrever(1);
    }
}
