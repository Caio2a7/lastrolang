// Problema 2: Crie um programa que leia uma quantidade desconhecida de números e informe
// quantos deles estão nos seguintes intervalos fechados: [0, 25], [26, 50], [51, 75]
// e [76, 100]. A entrada de dados deve terminar quando for lido um número negativo.

proc main() {
    int num := 0;
    int faixa_0_25 := 0;
    int faixa_26_50 := 0;
    int faixa_51_75 := 0;
    int faixa_76_100 := 0;

    ler(ref num);
    while (num >= 0) {
        if (num <= 25) {
            faixa_0_25 := faixa_0_25 + 1;
        } else if (num <= 50) {
            faixa_26_50 := faixa_26_50 + 1;
        } else if (num <= 75) {
            faixa_51_75 := faixa_51_75 + 1;
        } else if (num <= 100) {
            faixa_76_100 := faixa_76_100 + 1;
        }
        ler(ref num);
    }

    escrever("[0, 25]");
    escrever(faixa_0_25);
    escrever("[26, 50]");
    escrever(faixa_26_50);
    escrever("[51, 75]");
    escrever(faixa_51_75);
    escrever("[76, 100]");
    escrever(faixa_76_100);
}
